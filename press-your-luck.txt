JavaScript// Arrays of skit files (GIFs or MP4s) - populate with your generated assets
const randomSkits = [
  '/assets/whammies/disco.gif',
  '/assets/whammies/pie.gif',
  // Add all 15...
];
const winSkits = [
  '/assets/whammies/bomb-backfire.gif',
  // Add 5...
];
const loseSkits = [
  '/assets/whammies/victory-boom.gif',
  // Add 5...
];

// Function to show a skit
function showSkit(type = 'random') {
  const popup = document.getElementById('whammy-popup');
  const skitElement = popup.querySelector('.whammy-skit');
  const border = document.getElementById('game-border');
  let skitsArray = randomSkits;
  if (type === 'win') skitsArray = winSkits;
  if (type === 'lose') skitsArray = loseSkits;
  
  const randomSkit = skitsArray[Math.floor(Math.random() * skitsArray.length)];
  skitElement.innerHTML = `<img src="${randomSkit}" alt="Whammy Skit">`; // Or <video> for MP4
  
  document.body.classList.add('flashing');
  border.style.display = 'block';
  popup.style.display = 'block';
  
  // Auto-hide after 10s if no interaction
  const timeout = setTimeout(hideSkit, 10000);
  
  // Interrupt on user action
  const interrupt = () => {
    clearTimeout(timeout);
    hideSkit();
    window.removeEventListener('click', interrupt);
    window.removeEventListener('keydown', interrupt);
    window.removeEventListener('scroll', interrupt);
  };
  window.addEventListener('click', interrupt);
  window.addEventListener('keydown', interrupt);
  window.addEventListener('scroll', interrupt);
}

function hideSkit() {
  const popup = document.getElementById('whammy-popup');
  const border = document.getElementById('game-border');
  popup.classList.add('fade-out');
  border.classList.add('fade-out');
  document.body.classList.remove('flashing');
  setTimeout(() => {
    popup.style.display = 'none';
    border.style.display = 'none';
    popup.classList.remove('fade-out');
    border.classList.remove('fade-out');
  }, 1000);
}

// Dynamically position border icons (call on load)
function positionIcons() {
  const icons = document.querySelectorAll('.border-icon');
  const angleStep = 360 / icons.length;
  icons.forEach((icon, i) => {
    const angle = i * angleStep;
    icon.style.transform = `rotate(${angle}deg) translate( calc(50vw - 25px), calc(50vh - 25px) )`; // Adjust for screen size
  });
}
